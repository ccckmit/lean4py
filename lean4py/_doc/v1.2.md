### 1. ✅ 扩展 `optimization.py`: 共轭梯度法

```python
def conjugate_gradient(
    A: Callable[[List[float]], List[float]],  # Linear operator
    b: List[float],
    x0: Optional[List[float]] = None,
    max_iter: int = 1000,
    tol: float = 1e-6
) -> List[float]:
    """Conjugate Gradient method for solving Ax = b.
    
    Args:
        A: Function that computes A @ x (linear operator)
        b: Right-hand side vector
        x0: Initial guess (zeros if None)
        max_iter: Maximum iterations
        tol: Convergence tolerance
        
    Returns:
        Solution vector x
    """
    n = len(b)
    if x0 is None:
        x = [0.0] * n
    else:
        x = x0[:]
    
    r = [b[i] - A(x)[i] for i in range(n)]  # Residual
    p = r[:]  # Initial search direction
    
    for _ in range(min(max_iter, n)):
        Ap = A(p)
        alpha = sum(r[i]**2 for i in range(n)) / sum(p[i]*Ap[i] for i in range(n))
        x = [x[i] + alpha * p[i] for i in range(n)]
        r_new = [r[i] - alpha * Ap[i] for i in range(n)]
        
        if (sum(r_new[i]**2 for i in range(n)))**0.5 < tol:
            return x
        
        beta = sum(r_new[i]**2 for i in range(n)) / sum(r[i]**2 for i in range(n))
        p = [r_new[i] + beta * p[i] for i in range(n)]
        r = r_new
    
    return x
```

def anova_one_way(groups: List[List[float]]) -> Tuple[float, float, float]:
    """One-way ANOVA.
    
    Args:
        groups: List of samples
        
    Returns:
        (F_statistic, p_value, eta_squared)
    """
    k = len(groups)  # Number of groups
    n_total = sum(len(g) for g in groups)
    
    if k < 2 or n_total <= k:
        return (0.0, 1.0, 0.0)
    
    # Group means and overall mean
    group_means = [mean(g) for g in groups]
    overall_mean = mean([x for g in groups for x in g])
    
    # Between-group SS
    SS_between = sum(len(g) * (group_means[i] - overall_mean)**2 for i, g in enumerate(groups))
    df_between = k - 1
    
    # Within-group SS
    SS_within = sum(sum((x - group_means[i])**2 for x in g) for i, g in enumerate(groups))
    df_within = n_total - k
    
    # F statistic
    if SS_within == 0:
        return (float('inf'), 0.0, 1.0)
    
    MS_between = SS_between / df_between
    MS_within = SS_within / df_within
    F = MS_between / MS_within
    
    # Approximate p-value
    import math
    p = 1 - 0.5 * (1 + math.erf(F**0.5 / 2**0.5))
    if p > 0.5:
        p = 2 * (1 - p)
    
    # Effect size (eta-squared)
    eta_sq = SS_between / (SS_between + SS_within)
    
    return (F, p, eta_sq)
```

def adaptive_simpson(f: Callable[[float], float], a: float, b: float, 
                     tol: float = 1e-6, max_depth: int = 20) -> float:
    """Adaptive Simpson's rule for numerical integration.
    
    Args:
        f: Function to integrate
        a, b: Integration bounds
        tol: Error tolerance
        max_depth: Maximum recursion depth
        
    Returns:
        Integral approximation
    """
    def _simpson(f, a, b):
        c = (a + b) / 2
        h = (b - a) / 6
        return h * (f(a) + 4*f(c) + f(b))
    
    def _recursive(f, a, b, S, tol, depth):
        c = (a + b) / 2
        S_left = _simpson(f, a, c)
        S_right = _simpson(f, c, b)
        if depth <= 0 or abs(S_left + S_right - S) < 15 * tol:
            return S_left + S_right + (S_left + S_right - S) / 15
        return (_recursive(f, a, c, S_left, tol/2, depth-1) +
                _recursive(f, c, b, S_right, tol/2, depth-1))
    
    S0 = _simpson(f, a, b)
    return _recursive(f, a, b, S0, tol, max_depth)
```
