### 1. ✅ 扩展 `optimization.py`: 惩罚函数法

```python
def penalty_method(
    f: Callable[[List[float]], float],
    inequalities: List[Tuple[Callable[[List[float]], float], float]],
    x0: List[float],
    mu: float = 10.0,
    max_iter: int = 100
) -> Tuple[List[float], float]:
    """Penalty method for inequality-constrained optimization.
    
    Minimize f(x) subject to g_i(x) <= 0 for all i.
    
    Args:
        f: Objective function
        inequalities: List of (g_i, max_violation) pairs
        x0: Initial guess
        mu: Penalty parameter
        max_iter: Maximum iterations
        
    Returns:
        (x_opt, f_opt)
    """
    x = x0[:]
    n = len(x0)
    
    for _ in range(max_iter):
        # Augmented objective: f(x) + (mu/2) * Σ max(0, g_i(x))^2
        def augmented(x_val):
            total = f(x_val)
            for g, _ in inequalities:
                violation = max(0, g(x_val))
                total += (mu / 2) * violation ** 2
            return total
        
        # Simple gradient descent on augmented function
        h = 1e-5
        grad = [(augmented([x[j] + (h if j==i else x[j]) for j in range(n)]) - 
                  augmented(x)) / h
                  for i in range(n)]
        
        learning_rate = 0.01
        x = [x[j] - learning_rate * grad[j] for j in range(n)]
        
        # Check convergence
        max_violation = max(max(0, g(x)) for g, _ in inequalities)
        if max_violation < 1e-6:
            break
        
        mu *= 2  # Increase penalty
    
    return x, f(x)
```

### 2. ✅ 新建 `pde.py`: 偏微分方程求解

```python
import numpy as np
from typing import Callable, Tuple, List

def solve_heat_equation(
    L: float,
    T: float,
    u0: Callable[[float], float],
    alpha: float = 1.0,
    nx: int = 50,
    nt: int = 100
) -> Tuple[List[float], List[List[float]]]:
    """Solve heat equation u_t = alpha * u_xx with u(0,t)=u(L,t)=0.
    
    Returns:
        (x_grid, solution) where solution is 2D array
    """
    dx = L / (nx - 1)
    dt = T / nt
    r = alpha * dt / (dx ** 2)
    
    x = [i * dx for i in range(nx)]
    u = [u0(xi) for xi in x]
    
    for _ in range(nt):
        u_new = u[:]
        for i in range(1, nx-1):
            u_new[i] = u[i] + r * (u[i+1] - 2*u[i] + u[i-1])
        u = u_new
    
    return x, u
```

import inspect
import lean4py

def generate_api_doc():
    """Generate API reference from module docstrings."""
    modules = [
        'lean4py.logic',
        'lean4py.sets',
        'lean4py.algebra',
        # ... add all modules
    ]
    
    with open('API_REFERENCE.md', 'w') as f:
        f.write('# lean4py API Reference\n\n')
        for module_name in modules:
            module = __import__(module_name, fromlist=[''])
            f.write(f'## {module_name}\n\n')
            for name, obj in inspect.getmembers(module):
                if callable(obj) and not name.startswith('_'):
                    f.write(f'### {name}\n')
                    if obj.__doc__:
                        f.write(f'{obj.__doc__}\n\n')
```

def moving_average(x: List[float], window: int = 3) -> List[float]:
    """Simple moving average."""
    if len(x) < window:
        return []
    return [sum(x[i:i+window]) / window for i in range(len(x) - window + 1)]

def autocovariance(x: List[float], lag: int = 1) -> float:
    """Autocovariance at given lag."""
    n = len(x)
    if n <= lag:
        return 0.0
    mean_x = sum(x) / n
    return sum((x[i] - mean_x) * (x[i-lag] - mean_x) for i in range(lag, n)) / n

def acf(x: List[float], max_lag: int = 10) -> List[float]:
    """Autocorrelation function."""
    n = len(x)
    if n < 2:
        return []
    acov = [autocovariance(x, lag) for lag in range(max_lag + 1)]
    var = acov[0]
    if var == 0:
        return [0.0] * (max_lag + 1)
    return [acov[lag] / var for lag in range(max_lag + 1)]
```
