```python
from typing import TypeVar, Generic, Callable, List, Set, Optional

T = TypeVar('T')
U = TypeVar('U')

class Morphism(Generic[T, U]):
    """Morphism (arrow) between objects in a category."""
    def __init__(self, domain: T, codomain: U, apply: Callable):
        self.domain = domain
        self.codomain = codomain
        self.apply = apply

    def __compose__(self, other: 'Morphism') -> 'Morphism':
        """Composition: g ∘ f"""
        if self.domain != other.codomain:
            raise ValueError("Incompatible morphisms for composition")
        return Morphism(other.domain, self.codomain,
                       lambda x: self.apply(other.apply(x)))

class Category(Generic[T]):
    """Category: objects and morphisms satisfying composition axioms."""
    def __init__(self, name: str):
        self.name = name
        self.objects: Set[T] = set()
        self.morphisms: List[Morphism] = []

    def add_object(self, obj: T):
        self.objects.add(obj)

    def add_morphism(self, m: Morphism):
        self.morphisms.append(m)

    def identity(self, obj: T) -> Morphism:
        """Identity morphism for object."""
        return Morphism(obj, obj, lambda x: x)

class Functor(Generic[T, U]):
    """Functor F: C → D between categories."""
    def __init__(self, source: Category, target: Category,
                 object_map: Callable[[T], U],
                 morphism_map: Callable[[Morphism], Morphism]):
        self.source = source
        self.target = target
        self.object_map = object_map
        self.morphism_map = morphism_map

    def __call__(self, obj: T) -> U:
        return self.object_map(obj)

class NaturalTransformation(Generic[T, U]):
    """Natural transformation η: F ⇒ G between functors."""
    def __init__(self, source_functor: Functor, target_functor: Functor,
                 components: Callable[[T], Morphism]):
        self.source = source_functor
        self.target = target_functor
        self.components = components
```

```python
class Subgroup(Group):
    """Subgroup generated by a set."""
    def __init__(self, parent: Group, generators: set):
        self.parent = parent
        self.generators = generators
        self.carrier = self._generate()

    def _generate(self) -> set:
        """Generate subgroup from generators."""
        if not self.generators:
            return {self.parent.identity}
        closure = set(self.generators)
        # Close under group operation and inverses
        return closure

    def is_normal(self) -> bool:
        """Check if subgroup is normal."""
        for g in self.parent.carrier:
            for h in self.carrier:
                if not self.parent.op(g, self.parent.op(h, self.parent.inv(g))) in self.carrier:
                    return False
        return True

class QuotientGroup(Group):
    """Quotient group G/H where H is normal."""
    def __init__(self, group: Group, normal_subgroup: Subgroup):
        # Coset space G/H
        self.group = group
        self.normal_subgroup = normal_subgroup

    def coset(self, g) -> set:
        """Left coset gH."""
        return {self.group.op(g, h) for h in self.normal_subgroup.carrier}

class GroupHomomorphism:
    """Group homomorphism φ: G → H."""
    def __init__(self, source: Group, target: Group,
                 map_func: Callable):
        self.source = source
        self.target = target
        self.map_func = map_func

    def __call__(self, g):
        return self.map_func(g)

    def kernel(self) -> Subgroup:
        """Kernel of homomorphism: {g | φ(g) = e}."""
        return Subgroup(self.source,
                       {g for g in self.source.carrier
                        if self.map_func(g) == self.target.identity})

    def image(self) -> Subgroup:
        """Image of homomorphism: φ(G)."""
        return Subgroup(self.target,
                       {self.map_func(g) for g in self.source.carrier})
```

### 5. 新建 `lattice.py`: 格论

```python
from typing import Callable, List, Set, Optional

class Lattice:
    """Lattice: partially ordered set with meet and join."""
    def __init__(self, carrier: set, leq: Callable[[Any, Any], bool]):
        self.carrier = carrier
        self.leq = leq  # Partial order

    def meet(self, a, b) -> Any:
        """Greatest lower bound (infimum)."""
        # a ∧ b = greatest element ≤ a, b
        candidates = {x for x in self.carrier
                     if self.leq(x, a) and self.leq(x, b)}
        if not candidates:
            return None
        return max(candidates, key=lambda x: len({y for y in candidates if self.leq(y, x)}))

    def join(self, a, b) -> Any:
        """Least upper bound (supremum)."""
        # a ∨ b = least element ≥ a, b
        candidates = {x for x in self.carrier
                     if self.leq(a, x) and self.leq(b, x)}
        if not candidates:
            return None
        return min(candidates, key=lambda x: len({y for y in candidates if self.leq(x, y)}))

    def is_distributive(self) -> bool:
        """Check distributive: a ∧ (b ∨ c) = (a ∧ b) ∨ (a ∧ c)"""
        for a in self.carrier:
            for b in self.carrier:
                for c in self.carrier:
                    lhs = self.meet(a, self.join(b, c))
                    rhs = self.join(self.meet(a, b), self.meet(a, c))
                    if lhs != rhs:
                        return False
        return True

class BoundedLattice(Lattice):
    """Lattice with top and bottom elements."""
    def __init__(self, carrier: set, leq: Callable):
        super().__init__(carrier, leq)
        self.top = None
        self.bottom = None

class BooleanAlgebra(BoundedLattice):
    """Boolean algebra: distributive complemented lattice."""
    def __init__(self, carrier: set, leq: Callable):
        super().__init__(carrier, leq)

    def complement(self, a) -> Any:
        """Complement ¬a."""
        # a ∨ ¬a = ⊤ and a ∧ ¬a = ⊥
        for x in self.carrier:
            if (self.join(a, x) == self.top and
                self.meet(a, x) == self.bottom):
                return x
        return None
```
