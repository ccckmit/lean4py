### 1. ✅ 新建 `optimization.py`

```python
from typing import Callable, Tuple, List, Optional

def gradient_descent(
    f: Callable[[float], float],
    x0: float,
    learning_rate: float = 0.01,
    max_iter: int = 1000,
    tol: float = 1e-6
) -> Tuple[float, float]:
    """Gradient descent optimization (numerical gradient).
    
    Args:
        f: Objective function
        x0: Initial point
        learning_rate: Step size
        max_iter: Maximum iterations
        tol: Convergence tolerance
        
    Returns:
        (x_opt, f_opt): Optimal point and value
    """
    x = x0
    h = 1e-5  # For numerical derivative
    
    for _ in range(max_iter):
        # Numerical gradient
        grad = (f(x + h) - f(x - h)) / (2 * h)
        x_new = x - learning_rate * grad
        
        if abs(f(x_new) - f(x)) < tol:
            return x_new, f(x_new)
        x = x_new
    
    return x, f(x)

def linear_programming(
    c: List[float],
    A: List[List[float]],
    b: List[float],
    method: str = 'simplex'
) -> Optional[Tuple[float, List[float]]]:
    """Linear programming: minimize c^T x subject to Ax <= b.
    
    Uses scipy if available, otherwise falls back to error.
    
    Returns:
        (optimal_value, optimal_x) or None if infeasible
    """
    try:
        import numpy as np
        from scipy.optimize import linprog
        result = linprog(c, A_ub=A, b_ub=b, method=method)
        if result.success:
            return result.fun, list(result.x)
        return None
    except ImportError:
        raise ImportError("scipy required for linear_programming")
```

```python
from collections import deque

def max_flow(g: Graph, source: Any, sink: Any) -> Tuple[float, Graph]:
    """Ford-Fulkerson algorithm for max flow.
    
    Returns:
        (max_flow_value, residual_graph)
    """
    # Build capacity graph
    capacity = {}
    for u in g.vertices:
        for v in g.neighbors(u):
            cap = g.adjacency.get(u, {}).get(v, {}).get('weight', 1.0)
            capacity[(u, v)] = cap
    
    # Residual graph
    residual = Graph(g.vertices)
    for u, v in capacity:
        residual.add_edge(u, v, capacity=capacity[(u, v)])
        if (v, u) not in capacity:
            residual.add_edge(v, u, capacity=0.0)
    
    flow = 0
    while True:
        # BFS to find augmenting path
        path, path_flow = _find_augmenting_path(residual, source, sink)
        if not path:
            break
        flow += path_flow
        # Update residual capacities
        for i in range(len(path) - 1):
            u, v = path[i], path[i+1]
            # Decrease forward capacity
            edge_data = residual.adjacency[u][v]
            edge_data['capacity'] -= path_flow
            # Increase backward capacity
            if v not in residual.adjacency:
                residual.add_edge(v, u, capacity=0.0)
            residual.adjacency[v][u]['capacity'] += path_flow
    
    return flow, residual

def min_cut(g: Graph, source: Any, sink: Any) -> Tuple[float, set, set]:
    """Min cut using max flow result.
    
    Returns:
        (cut_capacity, S, T) where S contains source
    """
    flow_value, residual = max_flow(g, source, sink)
    # Find reachable vertices from source in residual
    visited = set()
    queue = deque([source])
    while queue:
        u = queue.popleft()
        if u in visited:
            continue
        visited.add(u)
        for v in residual.neighbors(u):
            cap = residual.adjacency[u][v].get('capacity', 1.0)
            if cap > 0 and v not in visited:
                queue.append(v)
    S = visited
    T = set(g.vertices) - S
    return flow_value, S, T
```
