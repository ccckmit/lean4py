### 1. ✅ 扩展 `optimization.py`: 牛顿法

```python
def newton_method(
    f: Callable[[float], float],
    x0: float,
    max_iter: int = 100,
    tol: float = 1e-6
) -> Tuple[float, float]:
    """Newton's method for optimization (second-order).
    
    Args:
        f: Objective function
        x0: Initial point
        max_iter: Maximum iterations
        tol: Convergence tolerance
        
    Returns:
        (x_opt, f_opt): Optimal point and value
    """
    x = x0
    h = 1e-5
    
    for _ in range(max_iter):
        # First derivative
        f_prime = (f(x + h) - f(x - h)) / (2 * h)
        # Second derivative
        f_double = (f(x + h) - 2*f(x) + f(x - h)) / (h ** 2)
        
        if abs(f_double) < 1e-10:
            break
            
        x_new = x - f_prime / f_double
        
        if abs(x_new - x) < tol:
            return x_new, f(x_new)
        x = x_new
    
    return x, f(x)
```

### 2. ✅ 扩展 `statistics.py`: 假设检验

```python
def t_test_one_sample(data: List[float], mu0: float = 0.0) -> Tuple[float, float]:
    """One-sample t-test.
    
    Returns:
        (t_statistic, p_value)
    """
    n = len(data)
    if n < 2:
        return (0.0, 1.0)
    x_bar = mean(data)
    s = std_dev(data)
    if s == 0:
        return (float('inf'), 0.0) if x_bar != mu0 else (0.0, 1.0)
    t = (x_bar - mu0) / (s / (n ** 0.5))
    # Approximate p-value (two-tailed)
    # Using simple approximation for normal if n > 30
    import math
    if n > 30:
        p = 2 * (1 - 0.5 * (1 + math.erf(abs(t) / 2 ** 0.5)))
    else:
        # Very rough approximation
        p = 2 * (1 - 0.5 * (1 + math.erf(abs(t) / 2 ** 0.5)))
    return (t, p)
```

### 3. ✅ 新建 `symbolic.py`

```python
try:
    import sympy
    SYMPY_AVAILABLE = True
except ImportError:
    SYMPY_AVAILABLE = False

def symbolic_derivative(expr_str: str, var: str = 'x') -> str:
    """Compute symbolic derivative.
    
    Args:
        expr_str: Expression like "x**2 + 3*x"
        var: Variable to differentiate
        
    Returns:
        Derivative as string
    """
    if not SYMPY_AVAILABLE:
        raise ImportError("sympy required for symbolic computation")
    import sympy as sp
    x = sp.symbols(var)
    expr = sp.sympify(expr_str)
    deriv = sp.diff(expr, x)
    return str(deriv)

def symbolic_integral(expr_str: str, var: str = 'x') -> str:
    """Compute symbolic indefinite integral."""
    if not SYMPY_AVAILABLE:
        raise ImportError("sympy required for symbolic computation")
    import sympy as sp
    x = sp.symbols(var)
    expr = sp.sympify(expr_str)
    integral = sp.integrate(expr, x)
    return str(integral)
```
