from typing import Generic, TypeVar, Callable, List, Dict

T = TypeVar('T')

class Limit(Generic[T]):
    """Limit of a diagram in a category."""
    def __init__(self, diagram: List['Morphism'], universal_property: Callable):
        self.diagram = diagram
        self.universal_property = universal_property

class Colimit(Generic[T]):
    """Colimit (direct limit) of a diagram."""
    def __init__(self, diagram: List['Morphism'], universal_property: Callable):
        self.diagram = diagram
        self.universal_property = universal_property

class TerminalObject:
    """Terminal object: exactly one morphism from any object."""
    def __init__(self, obj):
        self.obj = obj

class InitialObject:
    """Initial object: exactly one morphism to any object."""
    def __init__(self, obj):
        self.obj = obj

class Product(Generic[T]):
    """Product of objects: universal property for projections."""
    def __init__(self, objects: List[T], projections: List['Morphism']):
        self.objects = objects
        self.projections = projections

class Coproduct(Generic[T]):
    """Coproduct (sum) of objects: universal property for injections."""
    def __init__(self, objects: List[T], injections: List['Morphism']):
        self.objects = objects
        self.injections = injections

class Monad:
    """Monad T = (T, η, μ) on a category C."""
    def __init__(self, endofunctor: 'Functor',
                 unit: 'NaturalTransformation',
                 multiplication: 'NaturalTransformation'):
        self.endofunctor = endofunctor
        self.unit = unit  # η: Id ⇒ T
        self.multiplication = multiplication  # μ: T² ⇒ T

    def kleisli_category(self) -> 'KleisliCategory':
        """Kleisli category of the monad."""
        return KleisliCategory(self)

class KleisliCategory:
    """Kleisli category: objects are C-objects, morphisms are Kleisli arrows."""
    def __init__(self, monad: Monad):
        self.monad = monad

class Comonad:
    """Comonad (co-monad): dual of monad."""
    def __init__(self, endofunctor: 'Functor',
                 counit: 'NaturalTransformation',
                 comultiplication: 'NaturalTransformation'):
        self.endofunctor = endofunctor
        self.counit = counit  # ε: T ⇒ Id
        self.comultiplication = comultiplication  # δ: T ⇒ T²

class YonedaEmbedding:
    """Yoneda embedding: C → [C^op, Set]."""

        self.category = category

from typing import Callable, List, Dict, Tuple
import math

class GroupRepresentation:
    """Representation ρ: G → GL(V) of group G on vector space V."""
    def __init__(self, group: 'Group', dimension: int,
                 representation_map: Callable):
        self.group = group
        self.dimension = dimension
        self.representation_map = representation_map  # g → matrix

    def __call__(self, g) -> List[List[float]]:
        """Get matrix representation of group element g."""
        return self.representation_map(g)

class RepresentationHomomorphism:
    """Intertwining operator between representations."""
    def __init__(self, source: 'GroupRepresentation',
                 target: 'GroupRepresentation',
                 operator: List[List[float]]):
        self.source = source
        self.target = target
        self.operator = operator  # A such that ρ₂(g)A = Aρ₁(g)

    def is_intertwining(self) -> bool:
        """Check if A satisfies ρ₂(g)A = Aρ₁(g) for all g."""
        pass

class Character:
    """Character of a representation: χ(g) = Tr(ρ(g))."""
    def __init__(self, representation: GroupRepresentation):
        self.representation = representation

    def __call__(self, g) -> float:
        """Trace of representation matrix."""
        mat = self.representation(g)
        return sum(mat[i][i] for i in range(len(mat)))

    def inner_product(self, other: 'Character') -> float:
        """⟨χ, ψ⟩ = (1/|G|) Σ_g χ(g)ψ(g)̄."""
        pass

class IrreducibleRepresentation:
    """Irreducible representation (no proper invariant subspaces)."""
    def __init__(self, representation: GroupRepresentation):
