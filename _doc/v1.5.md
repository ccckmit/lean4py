```python
def augmented_lagrange(
    f: Callable[[List[float]], float],
    eq_constraints: List[Tuple[Callable[[List[float]], float], float]],
    ineq_constraints: List[Tuple[Callable[[List[float]], float], float]],
    x0: List[float],
    max_iter: int = 100,
    tol: float = 1e-6
) -> Tuple[List[float], float, List[float], List[float]]:
    """Augmented Lagrange method.
    
    Minimize f(x) subject to:
        g_i(x) = eq_target_i  (equality)
        h_j(x) <= ineq_target_j  (inequality)
    
    Returns:
        (x_opt, f_opt, lambda_opt, mu_opt)
    """
    x = x0[:]
    n = len(x0)
    m_eq = len(eq_constraints)
    m_ineq = len(ineq_constraints)
    
    lam = [0.0] * m_eq  # Lagrange multipliers for equality
    mu = [0.0] * m_ineq  # Lagrange multipliers for inequality
    rho = 1.0  # Penalty parameter
    
    for _ in range(max_iter):
        # Minimize augmented Lagrangian
        def L_aug(x_val):
            total = f(x_val)
            # Equality constraints
            for g, target in eq_constraints:
                violation = g(x_val) - target
                total += lam[i] * violation + (rho/2) * violation**2
            # Inequality constraints (with mu >= 0)
            for j, (h, target) in enumerate(ineq_constraints):
                violation = max(0, h(x_val) - target)
                total += mu[j] * violation + (rho/2) * violation**2
            return total
        
        # Simple gradient descent on L_aug
        h = 1e-5
        grad = [(L_aug([x[k] + (h if k==i else x[k]) for k in range(n)]) - L_aug(x)) / h
                for i in range(n)]
        learning_rate = 0.01
        x = [x[i] - learning_rate * grad[i] for i in range(n)]
        
        # Update multipliers
        for i, (g, target) in enumerate(eq_constraints):
            lam[i] += rho * (g(x) - target)
        
        for j, (h, target) in enumerate(ineq_constraints):
            violation = max(0, h(x) - target)
            mu[j] = max(0, mu[j] + rho * violation)
        
        # Check convergence
        eq_violations = [abs(g(x) - target) for g, target in eq_constraints]
        ineq_violations = [max(0, h(x) - target) for h, target in ineq_constraints]
        max_viol = max(eq_violations + ineq_violations)
        
        if max_viol < tol:
            break
        
        rho *= 1.5  # Increase penalty
    
    return x, f(x), lam, mu
```

```python
def linear_regression_diagnostics(x: List[float], y: List[float]) -> dict:
    """Linear regression with diagnostics.
    
    Returns:
        dict with: slope, intercept, r_squared, residuals, fitted_values
    """
    if len(x) != len(y) or len(x) < 2:
        return {}
    
    n = len(x)
    x_mean, y_mean = mean(x), mean(y)
    
    # Compute slope and intercept
    sxx = sum((xi - x_mean)**2 for xi in x)
    sxy = sum((x[i] - x_mean) * (y[i] - y_mean) for i in range(n))
    
    if sxx == 0:
        return {}
    
    slope = sxy / sxx
    intercept = y_mean - slope * x_mean
    
    # Compute fitted values and residuals
    fitted = [intercept + slope * xi for xi in x]
    residuals = [y[i] - fitted[i] for i in range(n)]
    
    # R-squared
    ss_res = sum(r**2 for r in residuals)
    ss_tot = sum((yi - y_mean)**2 for yi in y)
    r_squared = 1 - (ss_res / ss_tot) if ss_tot > 0 else 0.0
    
    return {
        'slope': slope,
        'intercept': intercept,
        'r_squared': r_squared,
        'residuals': residuals,
        'fitted_values': fitted
    }
```

```python
def solve_laplace_equation(
    Lx: float, Ly: float,
    nx: int = 50, ny: int = 50,
    max_iter: int = 1000, tol: float = 1e-6
) -> List[List[float]]:
    """Solve Laplace equation ∇²u = 0 with u=0 on boundary.
    
    Uses Jacobi iteration.
    """
    dx = Lx / (nx - 1)
    dy = Ly / (ny - 1)
    
    u = [[0.0 for _ in range(ny)] for _ in range(nx)]
    
    for _ in range(max_iter):
        u_new = [row[:] for row in u]
        max_diff = 0.0
        
        for i in range(1, nx-1):
            for j in range(1, ny-1):
                u_new[i][j] = 0.25 * (u[i+1][j] + u[i-1][j] + u[i][j+1] + u[i][j-1])
                max_diff = max(max_diff, abs(u_new[i][j] - u[i][j]))
        
        u = u_new
        if max_diff < tol:
            break
    
    return u


def solve_poisson_equation(
    Lx: float, Ly: float,
    source: Callable[[float, float], float],
    nx: int = 50, ny: int = 50,
    max_iter: int = 1000, tol: float = 1e-6
) -> List[List[float]]:
    """Solve Poisson equation ∇²u = f with u=0 on boundary."""
    dx = Lx / (nx - 1)
    dy = Ly / (ny - 1)
    dx2, dy2 = dx**2, dy**2
    
    u = [[0.0 for _ in range(ny)] for _ in range(nx)]
    
    for _ in range(max_iter):
        u_new = [row[:] for row in u]
        max_diff = 0.0
        
        for i in range(1, nx-1):
            for j in range(1, ny-1):
                x_val = i * dx
                y_val = j * dy
                f_val = source(x_val, y_val)
                u_new[i][j] = (dy2 * (u[i+1][j] + u[i-1][j]) +
                            dx2 * (u[i][j+1] + u[i][j-1]) -
                            dx2 * dy2 * f_val) / (2 * (dx2 + dy2))
                max_diff = max(max_diff, abs(u_new[i][j] - u[i][j]))
        
        u = u_new
        if max_diff < tol:
            break
    
    return u
```

```python
from typing import List, Tuple

def linear_regression_ml(x: List[List[float]], y: List[float]) -> Tuple[List[float], float]:
    """Linear regression using least squares.
    
    Args:
        x: Input features (list of feature vectors)
        y: Target values
        
    Returns:
        (coefficients, intercept)
    """
    if not x or not y:
        return [], 0.0
    
    n = len(x)
    m = len(x[0])  # Number of features
    
    # Add bias term
    X = [[1.0] + row for row in x]  # Add intercept column
    
    # Normal equation: β = (X^T X)^(-1) X^T y
    # Simplified: use gradient descent
    learning_rate = 0.01
    max_iter = 1000
    tol = 1e-6
    
    beta = [0.0] * (m + 1)
    
    for _ in range(max_iter):
        # Compute gradient
        grad = [0.0] * (m + 1)
        for i in range(n):
            pred = sum(beta[j] * X[i][j] for j in range(m+1))
            error = pred - y[i]
            for j in range(m+1):
                grad[j] += error * X[i][j]
        
        # Update
        beta_new = [beta[j] - learning_rate * grad[j] / n for j in range(m+1)]
        
        if sum((beta_new[j] - beta[j])**2 for j in range(m+1)) ** 0.5 < tol:
            beta = beta_new
            break
        
        beta = beta_new
    
    intercept = beta[0]
    coefficients = beta[1:]
    return coefficients, intercept


def logistic_regression(
    x: List[List[float]], y: List[int],
    learning_rate: float = 0.01, max_iter: int = 1000
) -> List[float]:
    """Logistic regression for binary classification.
    
    Args:
        x: Input features
        y: Binary labels (0 or 1)
        
    Returns:
        coefficients (including intercept)
    """
    n = len(x)
    m = len(x[0]) if x else 0
    
    # Add bias term
    X = [[1.0] + row for row in x]
    beta = [0.0] * (m + 1)
    
    for _ in range(max_iter):
        # Compute gradient of log-likelihood
        grad = [0.0] * (m + 1)
        for i in range(n):
            z = sum(beta[j] * X[i][j] for j in range(m+1))
            p = 1 / (1 + math.exp(-z))
            error = y[i] - p
            for j in range(m+1):
                grad[j] += error * X[i][j]
        
        beta = [beta[j] + learning_rate * grad[j] / n for j in range(m+1)]
    
    return beta
```
